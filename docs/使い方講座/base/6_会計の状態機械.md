元: `1_仕様調査.md` / `2_型定義.md`（型は `Types.swift`）/ `../siyou.md` A-4、A-19-2、A-19-3、A-19-8

`56_会計の状態を切り替える.md` が扱う部分だけを置きます。

**状態の「行き先」を UI に決めさせない**のがこのファイルの狙いです。
UI は「どのボタンが押されたか」だけを送り、`move(...)` が **行き先・確認・PIN** を返します。

## 0. A-4 の遷移図を、そのまま表にする

A-4 の図と表は、そのまま Swift の `switch` にできます。

| 状態 | 商品追加・取消 | 中断 | 在庫変動 |
| --- | --- | --- | --- |
| 待機中 | 商品追加で「スキャン中」へ遷移。取消の対象なし | **不可**（カートが空） | なし |
| スキャン中 | 可 | 可 | なし |
| 支払方法選択 | 不可（戻るで可） | 可 | なし |
| 現金入力中 | 不可 | 可 | なし |
| sats待ち | 不可（請求額固定） | 可 | なし |
| 併用現金入力中（sats確認済み） | 不可（請求額固定） | 可（**PIN必須**・確認あり・**記録を残す**） | なし |
| 完了 | 不可 | **不可** | 減算済み |

## 1. 遷移の「返り値」（A-19-9 #9）

```swift
/// 確認ダイアログに出すもの
struct ConfirmReason: Equatable {
    let title: String
    let message: String
}

/// 遷移の要求。**まだ遷移していない**。UI はこの返り値に従って動くだけ。
enum Move: Equatable {
    case allowed(SaleState)                       // このまま遷移してよい
    case needsConfirm(ConfirmReason, SaleState)   // 確認が済んだら `nextState` へ
    case needsPIN(ConfirmReason, SaleState)       // PIN ＋ 確認が済んだら `nextState` へ
    case rejected(String)                         // 遷移できない。理由をそのまま表示する

    var nextState: SaleState? {
        switch self {
        case .allowed(let s): return s
        case .needsConfirm(_, let s), .needsPIN(_, let s): return s
        case .rejected: return nil
        }
    }
    var reason: ConfirmReason? {
        switch self {
        case .allowed: return nil
        case .needsConfirm(let r, _), .needsPIN(let r, _): return r
        case .rejected(let m): return ConfirmReason(title: "操作できません", message: m)
        }
    }
}
```

> **`rejected` は「許可されない操作」のメッセージとして使う**ので、専用タイトルはありません。
> `reason` で「操作できません／在庫不足: …」と表示できます。

## 2. 会計の文脈

```swift
/// 状態以外に必要な値だけをまとめたもの。
/// これを1つ保存・復元すれば、会計の状態がまるごと復元できる（C-6 #2・#78、`19_lesson` の担当）
struct SaleContext: Codable, Equatable {
    var state: SaleState = .idle
    var cart: [CartLine] = []
    /// sats の入金が確認された時刻。**nil なら未確認**（A-19-2 の判別）
    var satsConfirmedAt: Date?
    /// 済ボタンで承認した請求sats額。承認後に復元しても変化しない（C-6 #78）
    var confirmedSats: Int?
}
```

## 3. 画面が送る「イベント」

UI は状態を直接書かず、**押されたボタン**だけを送ります。

```swift
enum SaleEvent: Equatable {
    case addFirstItem          // 待機中の最初の商品追加
    case addItem               // 以降の商品追加
    case removeItem            // 商品取消
    case toPaymentSelection    // 会計へ進む
    case backToScan            // 戻る（支払方法選択 → スキャン中）
    case chooseCash            // 現金
    case chooseSats            // sats払い
    case tapDone               // 済ボタン（sats待ち・承認前）
    case changeMethod          // 方法選び直し
    case suspend               // 会計中断
    case confirmCash(Int)      // 受取額確定（現金入力中）
    case confirmMixedCash(Int) // 受取額確定（併用現金入力中）
    case backToIdle            // 完了／中断のあと
}
```

## 4. 遷移表そのもの

```swift
/// 遷移を1つ判断する。**状態は書き換えない**（返り値の `nextState` を使う）
func move(_ e: SaleEvent, in ctx: SaleContext) -> Move {
    switch ctx.state {

    case .idle:
        switch e {
        case .addFirstItem: return .allowed(.scanning)
        case .addItem:      return .rejected("待機中は「スキャン中」です。商品をスキャンしてください")
        case .suspend:      return .rejected("中断できません（カートが空です）")   // A-4 の表
        default:            return .rejected("この状態ではできません")
        }

    case .scanning:
        switch e {
        case .addItem:  return .allowed(.scanning)
        // カートが空になったら待機中へ（A-4）
        case .removeItem: return .allowed(ctx.cart.count <= 1 ? .idle : .scanning)
        case .toPaymentSelection: return .allowed(.choosingPayment)
        case .suspend:  return .needsConfirm(suspendReason(in: ctx), .suspended)
        default:        return .rejected("この状態ではできません")
        }

    case .choosingPayment:
        switch e {
        case .backToScan: return .allowed(.scanning)
        case .chooseCash: return .allowed(.cashInput)
        case .chooseSats:
            // 遷移の条件「sats単価のある行が1行以上」（A-4、A-6-1）
            // ★ 判定は `canPayBySats()` 1 か所だけ（base/2）。ここで書き直さない
            guard ctx.cart.canPayBySats() else {
                return .rejected("sats単価のある商品がありません。現金でお支払いください")
            }
            return .allowed(.satsWaiting)
        case .suspend: return .needsConfirm(suspendReason(in: ctx), .suspended)
        default:       return .rejected("この状態では商品追加・取消はできません")   // A-4 の表
        }

    case .cashInput:
        switch e {
        case .confirmCash(let received):
            let due = ctx.cashDueJpy                         // 全行の円合計（§7）
            guard received >= due else {
                return .rejected("受取額が請求額（\(due)円）未満です")
            }
            return .allowed(.done)
        case .suspend: return .needsConfirm(suspendReason(in: ctx), .suspended)
        default:       return .rejected("この状態ではできません")
        }

    case .satsWaiting:
        switch e {
        case .tapDone:
            // 現金分の有無で遷移先が変わる（A-4、C-6 #23・#77）
            let t = ctx.cart.totals()
            return t.cashForSats == 0
                ? .needsConfirm(tapDoneReason(in: ctx), .done)
                : .needsConfirm(tapDoneReason(in: ctx), .mixedCashInput)
        case .changeMethod: return .allowed(.choosingPayment)   // A-19-8：2段の1つ目
        case .suspend:      return .needsConfirm(cancelReason(in: ctx), .suspended)
        default:            return .rejected("請求額が決まったので、方法や商品は変えられません")
        }

    case .mixedCashInput:
        switch e {
        case .confirmMixedCash(let received):
            let due = ctx.cashDueJpy                         // ★ 現金分だけ（§7）
            guard received >= due else {
                return .rejected("受取額が現金分（\(due)円）未満です")
            }
            return .allowed(.done)
        case .suspend:
            // sats確認済みなので PIN 必須・記録を残す（A-4、A-19-2）
            return .needsPIN(pinReason(), .suspended)
        default: return .rejected("請求額が決まったので、方法や商品は変えられません")
        }

    case .done:
        switch e {
        case .backToIdle: return .allowed(.idle)
        default:          return .rejected("完了した会計は中断できません")   // A-4 の表
        }

    case .suspended:
        switch e {
        case .backToIdle: return .allowed(.idle)
        default:          return .rejected("中断しました。待機中に戻ってください")
        }
    }
}
```

## 5. 確認ダイアログの中身

```swift
/// 中断の確認（A-4「中断時はカートの内容が破棄されるため、確認ダイアログを経る」）
func suspendReason(in ctx: SaleContext) -> ConfirmReason {
    let t = totalLines(ctx.cart)
    return ConfirmReason(
        title: "会計を中断しますか",
        message: "カートの内容が破棄されます。\n\(t.satsLine)\n\(t.cashLine)")
}

/// sats待ちのキャンセル（A-4「sats待ち → 中断: キャンセル（確認あり）」。PIN不要）
func cancelReason(in ctx: SaleContext) -> ConfirmReason {
    ConfirmReason(title: "sats払いをやめますか",
                  message: "カートの内容はそのまま残ります（中断すると破棄されます）。")
}

/// 済ボタンの確認（C-6 #23「確認ダイアログを経ずに完了しない」）
func tapDoneReason(in ctx: SaleContext) -> ConfirmReason {
    let t = ctx.cart.totals()
    var msg = "請求sats額 \(t.sats)sats。WoS で入金を確認してから押してください。"
    if t.cashForSats > 0 { msg += "\nこのあとに現金 \(t.cashForSats)円 を受け取ります。" }
    return ConfirmReason(title: "入金を確認しましたか", message: msg)
}

/// sats確認済みの中断（A-19-2：PIN必須・記録を残す）
func pinReason() -> ConfirmReason {
    ConfirmReason(title: "会計を中断しますか",
                  message: "sats の入金が確認済みのため、PIN が必要です。\n記録が残ります。")
}
```

> **`tapDoneReason` は必ず確認を挟む**ので、「確認ダイアログを経ずに完了する」事故が起きません。
> 実運用では、この確認で WoS 履歴を見る時間をとります（A-6-3）。

## 6. 確認・PIN が済んだあとに適用する

```swift
/// 確認・PIN が済んだ遷移を適用する。**`nextState` が nil のときは何もしない**
func apply(_ m: Move, to ctx: inout SaleContext) {
    guard let to = m.nextState else { return }
    ctx.state = to
}

/// 済ボタン承認時。**請求sats額をこのときだけ確定させる**（C-6 #78）
func markSatsConfirmed(_ ctx: inout SaleContext, at: Date) {
    ctx.satsConfirmedAt = at
    ctx.confirmedSats = ctx.cart.totals().sats
}
```

> `apply` は **状態を移すだけ**です。販売記録や在庫には触りません。
> それらは `3_標準ロジック.md` の `commitSale` / `suspendSale` の担当で、
> `state == .done` / `state == .suspended` になったときに呼びます（A-4 のトランザクション）。

## 7. 受取方法と現金で受け取る額は、導出できる

この2つは **`SaleContext` から計算できる**ので、**保存しなくてよい**（`19` の教訓）。

```swift
extension SaleContext {
    /// 会計全体の受取方法（A-13 の `method`）。まだ選んでいなければ nil。
    var method: PaymentMethod? {
        switch state {
        case .cashInput:      return .cash
        case .satsWaiting:    return cart.totals().cashForSats == 0 ? .lightning : .mixed
        case .mixedCashInput: return .mixed
        // 待機中〜支払方法選択はまだ選んでいない。
        // 中断・完了は復元しないので、判定しない（A-3、完了時は保存を消す）
        case .idle, .scanning, .choosingPayment, .suspended, .done: return nil
        }
    }

    /// この会計で**現金で受け取る額**。A-6-1 の決め대로決まる。
    /// - `sats待ち` / `併用現金入力中`: **現金分だけ**（sats分は WoS で受ける）
    /// - それ以外: 全行の円合計（全額現金）
    var cashDueJpy: Int {
        switch state {
        case .satsWaiting, .mixedCashInput: return cart.totals().cashForSats
        default:                            return cart.totals().cash
        }
    }
}
```

> **`cashDueJpy` を1か所に集約する**のは、`cashInput` に `totals.cash`、
> `mixedCashInput` に `totals.cashForSats` を書くと **sats 分まで二重に要求してしまう**ためです
> （C-6 #77）。UI も `move` も、この1つだけを見ます。

## 8. 保存・復元（`19` の担当）

`SaleContext` は `Codable` なので、**そのまま1つ**でファイルに保存できます。

```swift
/// 会計の文脈をファイルへ保存・復元する。**失敗は `throws` で外へ出す**（黙って捨てない）
struct SaleContextStore {
    let url: URL

    init(url: URL = FileManager.default.urls(for: .cachesDirectory, in: .userDomainMask)[0]
                          .appendingPathComponent("saleContext.json")) {
        self.url = url
    }

    /// 更新のたびに呼ぶ
    func save(_ ctx: SaleContext) throws {
        let data = try jsonEncoder.encode(ctx)      // base/2 の encoder（ISO 8601）
        try data.write(to: url, options: .atomic)   // ← 途中で落ちても壊れない
    }

    /// 起動時に呼ぶ。**「無い」は nil、「壊れている」は throw** で区別する
    func load() throws -> SaleContext? {
        guard FileManager.default.fileExists(atPath: url.path) else { return nil }
        let data = try Data(contentsOf: url)
        return try jsonDecoder.decode(SaleContext.self, from: data)
    }

    /// 完了・記録に残らない中断のあと（A-3）
    func clear() throws {
        guard FileManager.default.fileExists(atPath: url.path) else { return }
        try FileManager.default.removeItem(at: url)
    }
}
```

**設計の要点3つ**

| # | 决定的 | 理由 |
| --- | --- | --- |
| 1 | `.atomic` で書く | 一時ファイルに書いてから入れ替える。途中で強制終了しても前の内容が残る（A-3 の保持） |
| 2 | `try?` を使わない | `try?` は「保存が壊れた」を「保存が無い」に変えてしまう |
| 3 | `method` と `cashDueJpy` を保存しない | §7 のプロパティで計算できる（保存した値と計算値がズレない） |

> **保存するのは `SaleContext` そのもの**です。`SaleSession` のような別の struct を
> 作ると、「カート」と「請求額」という同じものが2つできてズレます（`19` の §わざと間違う①）。

## 9. 通しの確認

上のコード（`Types.swift` ＋ `3_標準ロジック.md` 相当）に続けて置きます。

```swift
// ════════ 通しの確認（会計の状態機械・A-4）════════

let now = Date(timeIntervalSince1970: 1_700_000_000)
let correctPIN = "1234"

func p0(_ s: String) { print("  " + s) }

/// Move を1行で表示する
func show(_ m: Move) -> String {
    switch m {
    case .allowed(let s):
        return "allowed(→\(s.rawValue))"
    case .needsConfirm(let r, let s):
        return "needsConfirm(\(r.title) → \(s.rawValue))"
    case .needsPIN(let r, let s):
        return "needsPIN(\(r.title) → \(s.rawValue))"
    case .rejected(let m):
        return "rejected(\(m))"
    }
}

/// イベントを1つ処理する。**`allowed` のときだけ状態を更新する**
@discardableResult
func tap(_ e: SaleEvent, in ctx: inout SaleContext) -> Move {
    let m = move(e, in: ctx)
    p0("\(ctx.state.rawValue) ＋ \(e) → " + show(m))
    if case .allowed = m { apply(m, to: &ctx) }
    return m
}

/// 確認ダイアログで「はい」を押した。PIN が要るときは PIN も確認する
func confirmOK(_ m: Move, in ctx: inout SaleContext, pin: String = "1234") {
    switch m {
    case .needsConfirm:
        apply(m, to: &ctx)
        p0("   → 確認ダイアログで「はい」→ \(ctx.state.rawValue)")
    case .needsPIN:
        guard pin == correctPIN else {
            p0("   → PIN が違う。**何も変わらない**（\(ctx.state.rawValue) のまま）")
            return
        }
        apply(m, to: &ctx)
        p0("   → PIN \(pin) を確認 → \(ctx.state.rawValue)")
    default:
        p0("   → 確認すべきではない（\(show(m))）")
    }
}

// ── 前提：sats単価あり1行・なし1行のカート ──
let st = Product(id: "ST-001", typeId: "sticker", name: "ステッカー まる",
                 priceJpy: 300, priceSats: 500, stock: 10, attr1: nil, attr2: nil, attr3: nil,
                 imageRef: nil, isPublic: true, updatedAt: nil)
let adhoc = CartLine.adhoc(id: "ADHOC-1", name: "野良（雑誌）", priceJpy: 400, priceSats: nil, qty: 1)
let stLine = CartLine(product: st, qty: 2)
let twoStickers = [stLine]

print("【1】待機中の禁止操作（A-4 の表「中断 不可」）")
var c = SaleContext()
p0("合計: " + String(describing: twoStickers.totals()))
tap(.suspend, in: &c)                  // rejected
tap(.addItem, in: &c)                  // rejected（最初の1個は addFirstItem）

print("【2】最初の商品追加 → スキャン中")
c.cart = twoStickers
tap(.addFirstItem, in: &c)

print("【3】商品追加と取消（取消で空なら待機中へ・A-4）")
c.cart = [stLine, adhoc]      // 2行にしておく
tap(.addItem, in: &c)
tap(.removeItem, in: &c)               // あと1行あるので、ここではスキャン中のまま
p0("   行数: \(c.cart.count)")
c.cart = []
tap(.removeItem, in: &c)               // カートが空 → 待機中へ

print("【4】会計へ進む → 支払方法選択")
c.cart = twoStickers
tap(.addFirstItem, in: &c)
p0("   合計: " + String(describing: c.cart.totals()))
tap(.toPaymentSelection, in: &c)

print("【5】sats単価のない行しかないときは sats払い不可（A-4 の遷移条件）")
var c2 = SaleContext(cart: [adhoc])
tap(.addFirstItem, in: &c2)
tap(.toPaymentSelection, in: &c2)
tap(.chooseSats, in: &c2)
tap(.chooseCash, in: &c2)

print("【6】sats待ち → 済ボタン（現金分0 → done）")
var c3 = SaleContext(cart: twoStickers)
_ = tap(.addFirstItem, in: &c3)
_ = tap(.toPaymentSelection, in: &c3)
_ = tap(.chooseSats, in: &c3)
p0("   合計: " + String(describing: c3.cart.totals()))
let mDone = tap(.tapDone, in: &c3)     // needsConfirm（確認ダイアログを経ずに完了しない・C-6 #23）
p0("   この時点ではまだ \(c3.state.rawValue)（確認していないので遷移していない）")
confirmOK(mDone, in: &c3)

print("【7】sats待ち → 方法選び直し → 現金（併用にならない・C-6 #83 / A-19-8）")
var c3b = SaleContext(cart: twoStickers)
_ = tap(.addFirstItem, in: &c3b)
_ = tap(.toPaymentSelection, in: &c3b)
_ = tap(.chooseSats, in: &c3b)
tap(.changeMethod, in: &c3b)           // → 支払方法選択（A-19-8 の2段の1つめ）
tap(.chooseCash, in: &c3b)             // → 現金入力中
tap(.confirmCash(600), in: &c3b)       // 全行の円合計 600 で確定
p0("   method = " + resolveMethod(chooseSats: false, cart: c3b.cart).rawValue + "（併用にならない）")

print("【8】現金入力中の確定（受取額不足は rejected）")
var c4 = SaleContext(cart: twoStickers)
_ = tap(.addFirstItem, in: &c4)
_ = tap(.toPaymentSelection, in: &c4)
_ = tap(.chooseCash, in: &c4)
tap(.confirmCash(500), in: &c4)         // 請求額600 未満
tap(.confirmCash(1000), in: &c4)        // 完了

print("【9】完了から中断はできない（A-4 の表）")
tap(.suspend, in: &c4)
tap(.backToIdle, in: &c4)

print("【10】併用会計：現金分あり → 済ボタンで mixedCashInput（C-6 #77）")
var c5 = SaleContext(cart: twoStickers + [adhoc])
_ = tap(.addFirstItem, in: &c5)
_ = tap(.toPaymentSelection, in: &c5)
_ = tap(.chooseSats, in: &c5)
p0("   合計: " + String(describing: c5.cart.totals()))
let mMixed = tap(.tapDone, in: &c5)    // 現金分があるので mixedCashInput へ
confirmOK(mMixed, in: &c5)
markSatsConfirmed(&c5, at: now)

print("【11】併用現金入力中の確定は「現金分だけ」（C-6 #77）")
tap(.confirmMixedCash(300), in: &c5)    // 現金分400 未満
tap(.confirmMixedCash(400), in: &c5)    // 完了
p0("   method = " + resolveMethod(chooseSats: true, cart: c5.cart).rawValue)

print("【12】中断の2通り（A-19-2）")
var c6 = SaleContext(cart: twoStickers)
_ = tap(.addFirstItem, in: &c6)
confirmOK(tap(.suspend, in: &c6), in: &c6)      // sats未確認 → needsConfirm（PIN不要）
tap(.backToIdle, in: &c6)

var c7 = SaleContext(state: .mixedCashInput, cart: twoStickers + [adhoc])
c7.satsConfirmedAt = now
let mPin = tap(.suspend, in: &c7)               // sats確認済み → needsPIN（記録を残す）
confirmOK(mPin, in: &c7, pin: "0000")           // PIN が違う → 何も変わらない
confirmOK(mPin, in: &c7)                        // 正しい PIN

print("【13】済ボタン承認後の請求sats額は固定される（C-6 #78）")
var c8 = SaleContext(state: .satsWaiting, cart: twoStickers)
markSatsConfirmed(&c8, at: now)
```

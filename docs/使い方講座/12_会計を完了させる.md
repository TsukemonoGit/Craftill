## 記録の形（sales / sale_items 相当）

```swift
struct Sale {
    let saleId: String
    var status: String          // completed / voided
    var method: String          // cash / lightning / mixed
    var totalJpy: Int?          // cash=円合計、mixed=現金分、lightning=nil
    var totalSats: Int?         // lightning・mixed の請求sats額、cash=nil
    var receivedJpy: Int?       // 受取額（cash・mixed）
    var changeJpy: Int?         // お釣り（cash・mixed）
    var confirmedAt: String?    // sats入金の確認時刻（lightning・mixed）
    var refunded: Bool          // 返金済みフラグ（sats分）
    var completedAt: String?    // 完了日時
}

struct SaleItem {
    let itemId: String
    let saleId: String
    let itemType: String        // registered / adhoc
    let typeId: String?         // registered のみ
    let productId: String?      // registered のみ
    let qty: Int
    let unitPriceJpy: Int       // 販売時点の円単価
    let unitPriceSats: Int?     // 販売時点のsats単価（nil なら現金で受け取る）
    let paidIn: String          // jpy / sats
    let nameSnapshot: String    // 販売時点の商品名
}
```

## 合計の計算

`sales.total_jpy` / `total_sats` を決めるのに使う。

```swift
struct Totals {
    var sats: Int = 0        // sats単価のある行のsats合計
    var cashForSats: Int = 0 // sats単価のない行の円合計（＝現金分）
    var cash: Int = 0        // 全行の円合計
}

func totals(_ cart: [CartLine]) -> Totals {
    var t = Totals()
    for line in cart {
        if let s = line.priceSats { t.sats += s * line.qty }
        else { t.cashForSats += line.priceJpy * line.qty }
        t.cash += line.priceJpy * line.qty
    }
    return t
}
```

## 大問: 会計1件を組み立てる

### 小問1: method を判定する

```swift
func resolveMethod(chooseSats: Bool, cart: [CartLine]) -> String {
    if !chooseSats { return "cash" }
    return totals(cart).cashForSats == 0 ? "lightning" : "mixed"
}
```

### 小問2: 明細の paid_in を決める

`cash` は全行 `jpy`。`lightning`・`mixed` は **sats単価がある行が `sats`、ない行が `jpy`**。

```swift
func paidIn(of line: CartLine, method: String) -> String {
    if method == "cash" { return "jpy" }
    return line.priceSats != nil ? "sats" : "jpy"
}
```

### 小問3: 会計1件を組み立てる

```swift
func buildSale(saleId: String,
               cart: [CartLine],
               chooseSats: Bool,
               receivedJpy: Int?,       // 現金を受け取ったときだけ渡す
               confirmedAt: String?,    // sats確認時刻（確認済みなら渡す）
               now: String) -> (sale: Sale, items: [SaleItem]) {

    let method = resolveMethod(chooseSats: chooseSats, cart: cart)
    let t = totals(cart)

    var sale = Sale(saleId: saleId, status: "completed", method: method,
                    totalJpy: nil, totalSats: nil, receivedJpy: nil,
                    changeJpy: nil, confirmedAt: nil, refunded: false,
                    completedAt: now)

    // ── 明細を作る ──
    var items: [SaleItem] = []
    for (n, line) in cart.enumerated() {
        items.append(SaleItem(
            itemId: String(format: "I-%03d", n + 1),
            saleId: saleId,
            itemType: line.isAdhoc ? "adhoc" : "registered",
            typeId: line.isAdhoc ? nil : line.typeId,
            productId: line.isAdhoc ? nil : line.id,
            qty: line.qty,
            unitPriceJpy: line.priceJpy,
            unitPriceSats: line.priceSats,
            paidIn: paidIn(of: line, method: method),
            nameSnapshot: line.name))
    }

    // ── method ごとに金額を入れる ──
    switch method {
    case "cash":
        sale.totalJpy = t.cash
    case "lightning":
        sale.totalSats = t.sats
        sale.confirmedAt = confirmedAt
    case "mixed":
        sale.totalSats = t.sats
        sale.totalJpy = t.cashForSats
        sale.confirmedAt = confirmedAt
    default:
        break
    }

    // ── 現金の受取額・お釣り（cash・mixed）──
    if method == "cash" || method == "mixed" {
        sale.receivedJpy = receivedJpy
        if let r = receivedJpy, let req = sale.totalJpy {
            sale.changeJpy = r - req
        }
    }

    return (sale, items)
}
```
- `for (n, line) in cart.enumerated()` … `n` は何番目か（0始まり）。`item_id` の連番に使う
- `String(format: "I-%03d", n + 1)` … `I-001` のようにゼロ埋め
- `switch method` で3通りに分ける。`total_jpy` は `cash`=円合計、`mixed`=現金分、`lightning`=なし

## 小問5: 気をつけること

- 金額は `sale.total_jpy`（現金で受け取る額）と `sale.total_sats`（請求sats額）を**別々に**持つ。円とsatsは合算しない。
- `item_type` は登録商品 `registered`、野良 `adhoc`。野良は `type_id`・`product_id` が **nil**。
- `name_snapshot` は「販売時点の名前」。後で商品名を変えても、記録は変わらない。
- 在庫の減算は**別の処理**（`8_在庫を減らす.md` の完了処理）。ここでは記録だけ。
- 本物のアプリでは、`sales`・`sale_items` の追加と在庫減算を、DBの**ひとつのトランザクション**で行う（A-4）。

## ポイント

- 会計1件は「`sales` 1行 ＋ `sale_items` のN行」でできている
- `paid_in` は `method` と「その行にsats単価があるか」で決まる
- 現金を受け取った会計（`cash`・`mixed`）だけ `received_jpy` / `change_jpy` を入れる
- sats払い（`lightning`・`mixed`）は `confirmed_at` と `total_sats` を入れる
- 野良は `adhoc` として記録し、在庫には触れない
